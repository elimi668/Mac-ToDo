"""TaskRepository：封装 Task 的 CRUD 与常用查询。短事务，每次操作独立 session。"""
from __future__ import annotations

from datetime import date, datetime

from sqlalchemy import select
from sqlalchemy.orm import Session, sessionmaker

from app.database.database import get_session
from app.models.task import Task


class _NoChangeType:
    """哨兵单例：区分 update() 中“不更新 deadline”与“传 None 清空 deadline”。"""
    _instance: "_NoChangeType | None" = None

    def __new__(cls) -> "_NoChangeType":
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __repr__(self) -> str:
        return "<no-change>"


NO_CHANGE = _NoChangeType()
"""传给 TaskRepository.update(deadline=...) 表示“不修改 deadline”。
传一个 date 表示设新值；传 None 表示清空。"""


class TaskRepository:
    """任务数据访问层。"""

    def __init__(self, session_factory: sessionmaker[Session] | None = None) -> None:
        # 允许测试注入；默认走模块级 SessionLocal
        self._session_factory = session_factory

    def _get_session(self) -> Session:
        if self._session_factory is not None:
            return self._session_factory()
        return get_session()

    # ---------- 新增 ----------
    def create(
        self,
        title: str,
        category: str = "生活",
        priority: int = 2,
        deadline: date | None = None,
    ) -> Task:
        title = title.strip()
        if not title:
            raise ValueError("任务标题不能为空")
        if priority not in (1, 2, 3):
            raise ValueError("priority 必须为 1/2/3")

        task = Task(
            title=title,
            category=category,
            priority=priority,
            deadline=deadline,
        )
        with self._get_session() as session:
            session.add(task)
            session.commit()
            session.refresh(task)
        return task

    # ---------- 查询 ----------
    def get_by_id(self, task_id: int) -> Task | None:
        with self._get_session() as session:
            return session.get(Task, task_id)

    def list_all(self) -> list[Task]:
        """未完成在前；同状态按截止日期升序、创建时间降序。"""
        with self._get_session() as session:
            stmt = select(Task).order_by(
                Task.completed.asc(),
                Task.deadline.asc().nulls_last(),
                Task.created_time.desc(),
            )
            return list(session.scalars(stmt))

    def filter_tasks(
        self,
        category: str | None = None,
        priority: int | None = None,
        completed: bool | None = None,
    ) -> list[Task]:
        """组合筛选。传 None 表示不筛选该维度。"""
        with self._get_session() as session:
            stmt = select(Task).order_by(
                Task.completed.asc(),
                Task.deadline.asc().nulls_last(),
                Task.created_time.desc(),
            )
            if category is not None:
                stmt = stmt.where(Task.category == category)
            if priority is not None:
                stmt = stmt.where(Task.priority == priority)
            if completed is not None:
                stmt = stmt.where(Task.completed == completed)
            return list(session.scalars(stmt))

    # ---------- 修改 ----------
    def update(
        self,
        task_id: int,
        title: str | None = None,
        category: str | None = None,
        priority: int | None = None,
        deadline: object = NO_CHANGE,
    ) -> Task | None:
        """更新任务字段。仅更新非 None 参数；
        deadline 用哨兵区分：不传=不改，传 date=改，传 None=清空。"""
        if priority is not None and priority not in (1, 2, 3):
            raise ValueError("priority 必须为 1/2/3")

        with self._get_session() as session:
            task = session.get(Task, task_id)
            if task is None:
                return None

            if title is not None:
                title = title.strip()
                if not title:
                    raise ValueError("任务标题不能为空")
                task.title = title
            if category is not None:
                task.category = category
            if priority is not None:
                task.priority = priority
            if deadline is not NO_CHANGE:
                task.deadline = deadline  # type: ignore[assignment]
            session.commit()
            session.refresh(task)
            return task

    def set_completed(self, task_id: int, completed: bool) -> Task | None:
        """切换完成状态，并记录/清除完成时间。"""
        with self._get_session() as session:
            task = session.get(Task, task_id)
            if task is None:
                return None
            task.completed = completed
            task.completed_time = datetime.now() if completed else None
            session.commit()
            session.refresh(task)
            return task

    # ---------- 删除 ----------
    def delete(self, task_id: int) -> bool:
        """删除任务。返回是否实际删除了一行。"""
        with self._get_session() as session:
            task = session.get(Task, task_id)
            if task is None:
                return False
            session.delete(task)
            session.commit()
            return True


    # ---------- 提醒 ----------
    def list_due_reminders(self, now: datetime, lead_minutes: int = 15) -> list[Task]:
        """查询需要发送截止提醒的任务。

        条件：未完成 + 未提醒 + deadline 在 now 到 now+lead_minutes 之间。
        """
        from datetime import timedelta

        end = now + timedelta(minutes=lead_minutes)
        with self._get_session() as session:
            stmt = select(Task).where(
                Task.completed == False,
                Task.reminded == False,
                Task.deadline.is_not(None),
                Task.deadline >= now.date(),
                # deadline 为 date 类型，仅按日期范围粗略过滤
            ).order_by(Task.deadline.asc(), Task.created_time.desc())
            tasks = list(session.scalars(stmt))
        # 精确过滤：deadline 当天落在 now 到 end 范围内
        return [t for t in tasks if t.deadline is not None and t.deadline == now.date()]

    def mark_reminded(self, task_id: int) -> None:
        """标记指定任务已发送过提醒。"""
        with self._get_session() as session:
            task = session.get(Task, task_id)
            if task is not None:
                task.reminded = True
                session.commit()

    # ---------- 日报周报 ----------
    def list_by_date_range(self, start: date, end: date) -> list[Task]:
        """查询截止日期在 [start, end] 范围内的任务（含两端）。"""
        with self._get_session() as session:
            stmt = select(Task).where(
                Task.deadline.is_not(None),
                Task.deadline >= start,
                Task.deadline <= end,
            ).order_by(Task.deadline.asc(), Task.created_time.desc())
            return list(session.scalars(stmt))

    def list_completed_in_range(self, start: datetime, end: datetime) -> list[Task]:
        """查询完成时间在 [start, end] 范围内的任务。"""
        with self._get_session() as session:
            stmt = select(Task).where(
                Task.completed == True,
                Task.completed_time.is_not(None),
                Task.completed_time >= start,
                Task.completed_time <= end,
            ).order_by(Task.completed_time.desc())
            return list(session.scalars(stmt))

    def list_created_in_range(self, start: datetime, end: datetime) -> list[Task]:
        """查询创建时间在 [start, end] 范围内的任务。"""
        with self._get_session() as session:
            stmt = select(Task).where(
                Task.created_time >= start,
                Task.created_time <= end,
            ).order_by(Task.created_time.desc())
            return list(session.scalars(stmt))

    # ---------- 数据库迁移 ----------
    @staticmethod
    def migrate() -> None:
        """检查并自动添加新字段（幂等）。"""
        from app.database.database import DB_FILE, _engine
        import sqlalchemy as sa

        if _engine is None:
            return
        try:
            with _engine.connect() as conn:
                inspector = sa.inspect(_engine)
                columns = {c["name"] for c in inspector.get_columns("tasks")}
                if "reminded" not in columns:
                    conn.execute(sa.text("ALTER TABLE tasks ADD COLUMN reminded BOOLEAN DEFAULT 0"))
                    conn.commit()
        except Exception:
            pass  # 表不存在等情况直接忽略
